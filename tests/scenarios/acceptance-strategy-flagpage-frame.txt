# the zoom's modal wait, which Escape gives up), the camera keys and the F-key slots (FUN_004705a0 0x5293f0: F1..F12 restore, Shift + F1..F10 save), H / N held (0x52940b / 0x529420: +-500000 a
# frame), the row counter set to 3 and "View Distance :%d" at (16, 32) (0x52944b).
# His path: Strategy Data, Mode > Module flags, then H, N, Insert, Escape, Home, A (forward), Left (strafe), Shift + F1, F1.
start-args C:/Users/Melanikus/source/AvP-builds-excluded/AvP-play-release/avp_rifs/Invasion.RIF
pictures on
state
expect-screen Construction
click-bar Mode > Strategy Data: Normal
expect-screen Strategy Data
click-bar Mode > Module flags
expect-screen Strategy Data
wait 300
expect-field camerasaved 1
# the View Distance line (16, 32)
expect-screen-text View Distance :2

# H held: the distance grows; N held: it shrinks
remember-field viewdistance
hold H 600
expect-field-changed viewdistance
remember-number screentext View_Distance_:
hold N 300
expect-number-moved screentext View_Distance_: -200 -1

# Insert: the zoom's wait ("Zoom... Choose centre"); Escape gives it up; Home: the default zoom
key Insert
expect-status Zoom... Choose centre
key Esc
expect-status Zoom cancelled
key Home
expect-status The zoom is back to the default level

# the camera keys: A forward (along z with the camera turned 0), Left strafes (along x)
camera -7764 1102 10527 0
wait 300
camera-note f0
hold A 400
camera-note f1
expect-travel-along f0 f1 z
hold Left 400
camera-note f2
expect-travel-along f1 f2 x

# the F-key slots: Shift + F1 saves the camera (slot 0, "Camera position saved"), F1 brings it back

wait 300
key F1 shift
wait 200
expect-field camerasaved 2
expect-field camerasaveddrawn 2
camera 0 0 0 0
wait 300
key F1
wait 300
expect-field-has camera 1000 -500 2000 1024
